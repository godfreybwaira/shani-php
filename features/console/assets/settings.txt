<?php

namespace {namespace}\{config_dir} {

    use shani\config\PathConfig;
    use shani\contracts\BasicConfiguration;
    use features\middleware\MiddlewareHandlerInterface;
    use {namespace}\{middleware_dir}\{middleware_name};
    use shani\launcher\App;

    final class {file_name} extends BasicConfiguration
    {

        private readonly PathConfig $pathConfig;
        private readonly {middleware_name} $middleware;

        public function __construct(App $app)
        {
            parent::__construct($app);
        }

        #[\Override]
        public function pathConfig(): PathConfig
        {
            return $this->pathConfig ??= new PathConfig(
                    mapper: $this->app->preference->mapper,
                    versionNumber: $this->app->preference->versionNumber,
                    homePath: '/{home_path}'
            );
        }

        #[\Override]
        public function isAsync(): bool
        {
            return $this->app->request->header()->getOne('X-Request-Mode') === 'async';
        }

        #[\Override]
        public function getMiddlewareHandler(): ?MiddlewareHandlerInterface
        {
            return $this->middleware ??= new {middleware_name}();
        }
    }

}
